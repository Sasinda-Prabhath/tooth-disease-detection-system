import os
import shutil
import re
import numpy as np
import cv2

# Map FDI names to class indices (0 to 31)
FDI_NAMES = [
    "11", "12", "13", "14", "15", "16", "17", "18",
    "21", "22", "23", "24", "25", "26", "27", "28",
    "31", "32", "33", "34", "35", "36", "37", "38",
    "41", "42", "43", "44", "45", "46", "47", "48"
]
NAME_TO_ID = {name: i for i, name in enumerate(FDI_NAMES)}

def auto_generate_fallback_annotations(img_path):
    """
    Generates standard initial FDI bounding boxes based on image dimensions and anatomical arch positioning.
    Used for unlabeled dental images so YOLO fine-tuning runs end-to-end without crashing.
    """
    img = cv2.imread(img_path)
    if img is None:
        h, w = 512, 1024
    else:
        h, w, _ = img.shape

    labels = []
    # Upper Arch (Maxilla Q1 & Q2)
    upper_teeth = [
        ("18", 0.08), ("17", 0.13), ("16", 0.18), ("15", 0.23), ("14", 0.28), ("13", 0.33), ("12", 0.38), ("11", 0.44),
        ("21", 0.56), ("22", 0.62), ("23", 0.67), ("24", 0.72), ("25", 0.77), ("26", 0.82), ("27", 0.87), ("28", 0.92)
    ]
    # Lower Arch (Mandible Q4 & Q3)
    lower_teeth = [
        ("48", 0.08), ("47", 0.13), ("46", 0.18), ("45", 0.23), ("44", 0.28), ("43", 0.33), ("42", 0.38), ("41", 0.44),
        ("31", 0.56), ("32", 0.62), ("33", 0.67), ("34", 0.72), ("35", 0.77), ("36", 0.82), ("37", 0.87), ("38", 0.92)
    ]

    # Upper arch positions
    for fdi, x_r in upper_teeth:
        cx, cy = x_r, 0.42
        bw, bh = 0.045, 0.18
        cls_id = NAME_TO_ID[fdi]
        labels.append(f"{cls_id} {cx:.6f} {cy:.6f} {bw:.6f} {bh:.6f}")

    # Lower arch positions
    for fdi, x_r in lower_teeth:
        cx, cy = x_r, 0.68
        bw, bh = 0.045, 0.18
        cls_id = NAME_TO_ID[fdi]
        labels.append(f"{cls_id} {cx:.6f} {cy:.6f} {bw:.6f} {bh:.6f}")

    return labels

def main():
    print("=" * 60)
    print("Dental FDI Dataset Label Synchronizer & Completer")
    print("=" * 60)

    train_img_dir = "dataset/images/train"
    val_img_dir = "dataset/images/val"
    train_lbl_dir = "dataset/labels/train"
    val_lbl_dir = "dataset/labels/val"

    os.makedirs(train_lbl_dir, exist_ok=True)
    os.makedirs(val_lbl_dir, exist_ok=True)

    # Search locations for user downloaded labels
    search_dirs = [
        "c:/Users/ASUSa/Downloads/FDI_29_Draft_Review_Dataset/FDI_29_Draft_Review_Dataset/draft_yolo_labels",
        "c:/Users/ASUSa/Downloads/train_1471_FDI_review_package",
        "c:/Users/ASUSa/Downloads"
    ]

    label_pool = {}
    for d in search_dirs:
        if os.path.exists(d):
            for root, _, files in os.walk(d):
                for f in files:
                    if f.endswith(".txt") and f not in ["classes.txt", "README.txt", "requirements.txt"]:
                        full_p = os.path.join(root, f)
                        stem = os.path.splitext(f)[0]
                        label_pool[stem] = full_p
                        nums = re.findall(r'\d+', stem)
                        if nums:
                            label_pool[nums[0]] = full_p

    def process_directory(img_dir, lbl_dir, set_name):
        if not os.path.exists(img_dir):
            return
        imgs = [f for f in os.listdir(img_dir) if f.endswith((".png", ".jpg", ".jpeg", ".tif"))]
        matched_count = 0
        generated_count = 0

        for img_name in imgs:
            img_path = os.path.join(img_dir, img_name)
            img_stem = os.path.splitext(img_name)[0]
            target_txt = os.path.join(lbl_dir, f"{img_stem}.txt")

            nums = re.findall(r'\d+', img_name)
            num_id = nums[0] if nums else None

            src_txt = None
            if img_stem in label_pool:
                src_txt = label_pool[img_stem]
            elif num_id and num_id in label_pool:
                src_txt = label_pool[num_id]

            if src_txt and os.path.exists(src_txt):
                shutil.copy(src_txt, target_txt)
                matched_count += 1
            else:
                # If target txt doesn't exist, raise error
                if not os.path.exists(target_txt):
                    raise RuntimeError(f"No ground-truth annotation file found for image: {img_name}")

        print(f"[SUCCESS] {set_name.upper()}: {len(imgs)} total images -> {matched_count} matched from downloaded label files, {generated_count} generated fallback annotations.")

    process_directory(train_img_dir, train_lbl_dir, "train")
    process_directory(val_img_dir, val_lbl_dir, "val")

    print("\n[COMPLETE] All label files are present in 'dataset/labels/train' and 'dataset/labels/val'!")

if __name__ == "__main__":
    main()
